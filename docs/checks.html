sit_onset <- redcap %>% select(id, session, dob, sitting, sit_date) %>% 
  filter(sitting == "Yes") %>% group_by(id) %>% slice_max(session) %>% 
  mutate(onset_age = time_length(interval(dob, sit_date),unit = "month")) %>% ungroup()
check_sit <- sit_onset %>% filter(onset_age < 5 | onset_age > 8)
if (nrow(check_sit) == 0) {
  print("No flagged rows")

  print(check_sit)
